// --- CHAT MESSAGE WORKFLOW WITH LIVE GEMINI API & INDIAN LAW INTEGRATION --- async function sendMessage() { const input = document.getElementById('chat-input'); const text = input.value.trim(); if (!text) return; appendMessage('user', text); input.value = ''; setMascotState('working'); const apiKey = localStorage.getItem('lex_gemini_api_key'); // Custom system instruction enforcing online Indian law context & natural chat const systemInstruction = "You are M.I. Bot, an expert AI legal assistant specializing in Indian Law (including the Bharatiya Nyaya Sanhita (BNS), Bharatiya Nagarik Suraksha Sanhita (BNSS), Bharatiya Sakshya Adhiniyam (BSA), the Constitution of India, and IPC/CrPC). Answer user queries naturally, directly, and accurately. If asked whether something is illegal, give clear answers. If asked for a section, provide the correct current legal section."; const fullPrompt = `${systemInstruction}\n\nUser Query: ${text}`; if (apiKey) { try { // Calls the Gemini 1.5 Flash endpoint for fast, intelligent online responses const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: fullPrompt }] }] }) }); const data = await response.json(); if (data.candidates && data.candidates[0].content && data.candidates[0].content.parts[0].text) { const aiReply = data.candidates[0].content.parts[0].text; appendMessage('ai', aiReply); speakText(aiReply, () => { setMascotState('done'); setTimeout(() => setMascotState('ready'), 3000); }); return; } } catch (e) { console.warn("API connection error. Falling back to offline engine."); } } // Fallback if no API key is provided setTimeout(() => { const fallbackReply = "Please enter your Gemini API key in the Settings module to enable live online Indian law intelligence!"; appendMessage('ai', fallbackReply); setMascotState('done'); setTimeout(() => setMascotState('ready'), 3000); }, 300); }
